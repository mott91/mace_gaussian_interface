```python
def calculator(nnp):

    if nnp == "mace_mp":
        from mace.calculators import mace_mp
        calc = mace_mp(model="large", device="cuda", default_dtype="float64", dispersion=False)
        return calc
    if nnp == "mace_off":
        from mace.calculators import mace_off
        calc = mace_off(model="large", device="cuda", default_dtype="float64", dispersion=False)
        return calc
    if nnp == "mace_omol":
        from mace.calculators import mace_omol
        calc = mace_omol(model="extra_large", device="cuda", default_dtype="float64", dispersion=False)
        return calc
    if nnp == "mace_anicc":
        from mace.calculators import mace_anicc
        calc = mace_anicc(device="cuda")
        return calc
```
