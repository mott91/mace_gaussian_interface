```python
import json
import os

def load_metrics(molecule):
    """Load metrics_summary.json for a molecule, return sorted comparisons."""
    path = os.path.join(
        os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
        f"analysis_results_harmonic/{molecule}/data/metrics_summary.json"
    )
    with open(path) as f:
        data = json.load(f)
    return sorted(data["comparisons"], key=lambda c: c["mae_freq"])


def parse_combo_name(name):
    """Split combo name into (energy_model, dipole_model)."""
    for suffix in ("_mace_ml", "_espaloma"):
        if name.endswith(suffix):
            return name[: -len(suffix)], suffix[1:]
    return name, "?"
```

def slide_results_table(prs):
    """Slide 8: Harmonic benchmark — water (8 combos) and aspirin (4 combos)."""
    water = load_metrics("water")
    aspirin = load_metrics("aspirin")

    def table_lines(combos, mol_label, ref_mae):
        lines = [
            (f"# Harmonic benchmark — {mol_label}", ACCENT),
            ("  Ref: B3LYP/6-31G(d,p)   |   sorted by frequency MAE", DIM),
            ("", DIM),
            ("  Energy model    Dipole      MAE(freq)   R²(freq)   R²(intens)", DIM),
            ("  " + "─" * 57, DIM),
        ]
        for c in combos:
            energy, dipole = parse_combo_name(c["name"])
            marker = "  ✓" if dipole == "mace_ml" else ""
            row = (
                f"  {energy:<14}  {dipole:<10}"
                f"  {c['mae_freq']:>5.0f} cm⁻¹"
                f"   {c['r2_freq']:.6f}"
                f"   {c['r2_intensity']:.2f}{marker}"
            )
            color = GREEN if c["mae_freq"] < ref_mae else (YELLOW if c["mae_freq"] < ref_mae * 5 else TEXT)
            lines.append((row, color))
        return lines

    water_lines = table_lines(water, "water (H₂O, 3 modes)", ref_mae=30)
    aspirin_lines = table_lines(aspirin, "aspirin (C₉H₈O₄, 51 modes)", ref_mae=20)

    # Water slide
    content_slide(prs, "$ cat results/water/metrics_summary.json | sort-by-mae", water_lines + [
        ("", DIM),
        ("  → Frequencies: all combos excellent (R² > 0.9999)", TEXT),
        ("  → Intensities: mace_ml ✓ consistently beats espaloma", GREEN),
    ])

    # Aspirin slide
    content_slide(prs, "$ cat results/aspirin/metrics_summary.json | sort-by-mae", aspirin_lines + [
        ("", DIM),
        ("  → mace_omol: best accuracy (MAE < 9 cm⁻¹)", GREEN),
        ("  → mace_mp: most speedup (~29×) but worse accuracy", YELLOW),
    ])

def slide_scaling(prs):
    """Slide N: Molecule size vs speedup — ASCII bar chart."""
    MAX_BAR = 20
    molecules = [
        ("water",   3,  1,  "~1×"),
        ("glycine", 10, 6,  "~7–10×"),
        ("aspirin", 21, 20, "~18–29×"),
    ]
    bar_lines = []
    for name, atoms, bar_val, label in molecules:
        bar = "=" * bar_val + " " * (MAX_BAR - bar_val)
        color = GREEN if bar_val >= 10 else (YELLOW if bar_val >= 3 else DIM)
        bar_lines.append((
            f"  {name:<10}  ({atoms:>2} atoms)  |{bar}|  {label}",
            color,
        ))

    content_slide(prs, "$ python benchmark.py --scaling", [
        ("# ML speedup scales with molecule size", ACCENT),
