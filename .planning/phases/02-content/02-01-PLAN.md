These helpers read from `comparison_results/{molecule}/{combo}/results.json` — one file per combo directory — and aggregate into a list sorted by r2_freq descending (highest R² = best = shown first).

```python
import json
import os


def load_combo_metrics(molecule):
    """Load per-combo results.json from comparison_results/ and return list sorted by r2_freq descending.

    Data source: comparison_results/{molecule}/{combo}/results.json (one file per combo directory).
    Sorted by r2_freq descending so the best-performing combo appears first.
    """
    base = os.path.join(
        os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
        "comparison_results",
        molecule,
    )
    combos = []
    if not os.path.isdir(base):
        return combos
    for combo_dir in sorted(os.listdir(base)):
        results_path = os.path.join(base, combo_dir, "results.json")
        if not os.path.isfile(results_path):
            continue
        with open(results_path) as f:
            data = json.load(f)
        combos.append({
            "name": combo_dir,
            "mae_freq": data.get("mae_freq", float("inf")),
            "r2_freq": data.get("r2_freq", 0.0),
            "r2_intensity": data.get("r2_intensity", 0.0),
            "speedup": data.get("speedup"),
        })
    # Sort by r2_freq descending (highest R² = best = shown first)
    combos.sort(key=lambda c: c["r2_freq"], reverse=True)
    return combos


def parse_combo_name(name):
    """Split combo name into (energy_model, dipole_model)."""
    for suffix in ("_mace_ml", "_espaloma"):
        if name.endswith(suffix):
            return name[: -len(suffix)], suffix[1:]
    return name, "?"
```

These two functions have no side effects and do not render slides. They will be called by slide_results_table() in Task 2b.

def slide_results_table(prs):
    """Slides 8–9: Harmonic benchmark — water (8 combos) and aspirin (4 combos).

    Rows sorted by R²(freq) descending. Uniform TEXT color — no per-row color logic.
    Data loaded from comparison_results/{molecule}/{combo}/results.json.
    """
    water = load_combo_metrics("water")
    aspirin = load_combo_metrics("aspirin")

    def table_lines(combos, mol_label):
        lines = [
            (f"# Harmonic benchmark — {mol_label}", ACCENT),
            ("  Ref: B3LYP/6-31G(d,p)   |   sorted by R²(freq) descending", DIM),
            ("", DIM),
            ("  Energy model    Dipole      R²(freq)    MAE(freq)   R²(intens)", DIM),
            ("  " + "─" * 60, DIM),

        for c in combos:
            energy, dipole = parse_combo_name(c["name"])

                f"  {energy:<14}  {dipole:<10}"

                f"   {c['mae_freq']:>6.1f} cm⁻¹"
                f"   {c['r2_intensity']:.2f}"
