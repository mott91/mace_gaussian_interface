   **Before** (inside run_frequency_calculation):
   ```python
   proc = subprocess.Popen(["g16", gjf_temp])
   counter = 0
   calc_start_time = time.time()
   with zmq_server(".ipc_file") as socket:
       while not is_calc_finished(proc, socket):
           elapsed = time.time() - calc_start_time
           if elapsed > GAUSSIAN_TIMEOUT_SECONDS:
               proc.kill()
               proc.wait()
               raise TimeoutError(...)
           logger.info(f"Calculation step {counter}")
           counter += 1
           msg = socket.recv_string()
           try:
               run_next_calculation(mol, msg, calc, dipole_method=dipole_calculator_name, calculate_derivatives=True)
               socket.send_string("ready")
           except Exception as e:
               logger.error(f"Calculation step failed: {e}")
               socket.send_string("error")
               return False
   ```

   **After** (replace the entire block with):
   ```python
   def _on_request(msg: str) -> str:
       run_next_calculation(
           mol,
           msg,
           calc,
           dipole_method=dipole_calculator_name,
           calculate_derivatives=True,
       )
       return "ready"

   run_gaussian_with_zmq(
       gjf_temp,
       on_request=_on_request,
       timeout_seconds=DEFAULT_TIMEOUT_SECONDS,
   )
   ```

   Note: the `try/except Exception: return False` that wrapped the old block is superseded by GaussianTimeoutError and GaussianRunError — the outer try/except in run_frequency_calculation() catches all exceptions already.
