    class _WellFormedChecker(HTMLParser):
        def __init__(self):
            super().__init__()
            self.stack = []

        def handle_starttag(self, tag, attrs):
            if tag not in ("br", "img", "meta", "link", "input", "hr"):

            if self.stack and self.stack[-1] == tag:
